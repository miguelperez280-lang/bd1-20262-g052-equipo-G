BEGIN;

/* ---------------------------------------------------------------------
   1. ADD COLUMN + ADD CONSTRAINT (agregar atributos y reglas)
   --------------------------------------------------------------------- */

-- 1.1 usuario: se agrega el teléfono de contacto con su regla de formato
ALTER TABLE usuario
    ADD COLUMN telefono VARCHAR(15);

ALTER TABLE usuario
    ADD CONSTRAINT ck_usuario_telefono
    CHECK (telefono ~ '^[0-9+]{7,15}$');

-- 1.2 producto: estado de disponibilidad del artículo
ALTER TABLE producto
    ADD COLUMN estado VARCHAR(15) NOT NULL DEFAULT 'DISPONIBLE';

ALTER TABLE producto
    ADD CONSTRAINT ck_producto_estado
    CHECK (estado IN ('DISPONIBLE', 'RESERVADO', 'VENDIDO'));

-- 1.3 evento: modalidad de realización
ALTER TABLE evento
    ADD COLUMN modalidad VARCHAR(10) NOT NULL DEFAULT 'PRESENCIAL';

ALTER TABLE evento
    ADD CONSTRAINT ck_evento_modalidad
    CHECK (modalidad IN ('PRESENCIAL', 'VIRTUAL', 'HIBRIDO'));

-- 1.4 publicacion: título opcional para la publicación
ALTER TABLE publicacion
    ADD COLUMN titulo VARCHAR(150);

-- 1.5 grupo: permite desactivar grupos sin borrarlos
ALTER TABLE grupo
    ADD COLUMN activo BOOLEAN NOT NULL DEFAULT TRUE;

-- 1.6 horario: aula o enlace donde se dicta la clase
ALTER TABLE horario
    ADD COLUMN aula VARCHAR(30);

/* ---------------------------------------------------------------------
   2. ALTER COLUMN (cambiar tipo, tamaño o valor por defecto)
   --------------------------------------------------------------------- */

-- 2.1 perfil: ampliar el tamaño de la biografía
ALTER TABLE perfil
    ALTER COLUMN biografia TYPE VARCHAR(1000);

-- 2.2 credencial: la fecha de emisión toma por defecto la fecha actual
ALTER TABLE credencial
    ALTER COLUMN fecha_emision SET DEFAULT CURRENT_DATE;

-- 2.3 servicio: ampliar el tamaño del nombre
ALTER TABLE servicio
    ALTER COLUMN nombre TYPE VARCHAR(150);

/* ---------------------------------------------------------------------
   3. RENAME COLUMN (consistencia de nomenclatura)
   --------------------------------------------------------------------- */

-- 3.1 comentario: nombre homogéneo con el resto de tablas (fecha_creacion)
ALTER TABLE comentario
    RENAME COLUMN fecha_comentario TO fecha_creacion;

-- 3.2 perfil_grupo: la imagen representa la portada del grupo
ALTER TABLE perfil_grupo
    RENAME COLUMN imagen_url TO portada_url;

/* ---------------------------------------------------------------------
   4. DROP CONSTRAINT / ADD CONSTRAINT (modificar una clave foránea)
   --------------------------------------------------------------------- */

-- 4.1 servicio: si el oferente se elimina, se eliminan sus servicios
ALTER TABLE servicio
    DROP CONSTRAINT fk_servicio_usuario_oferente;

ALTER TABLE servicio
    ADD CONSTRAINT fk_servicio_usuario_oferente
    FOREIGN KEY (id_usuario_oferente) REFERENCES usuario (id_usuario)
    ON UPDATE CASCADE ON DELETE CASCADE;

-- 4.2 usuario: el teléfono debe ser único entre usuarios
ALTER TABLE usuario
    ADD CONSTRAINT uq_usuario_telefono UNIQUE (telefono);

/* ---------------------------------------------------------------------
   5. CREATE INDEX (mejorar consultas frecuentes sobre claves foráneas)
   --------------------------------------------------------------------- */

CREATE INDEX ix_publicacion_usuario     ON publicacion (id_usuario);
CREATE INDEX ix_publicacion_fecha       ON publicacion (fecha_publicacion DESC);
CREATE INDEX ix_comentario_publicacion  ON comentario (id_publicacion);
CREATE INDEX ix_producto_tipo_producto  ON producto (id_tipo_producto);
CREATE INDEX ix_evento_fecha_inicio     ON evento (fecha_inicio);

/* ---------------------------------------------------------------------
   6. DROP COLUMN (eliminar un atributo que ya no se requiere)
   --------------------------------------------------------------------- */

-- 6.1 perfil: los intereses se gestionarán dentro de la biografía (2.1)
ALTER TABLE perfil
    DROP COLUMN intereses;

COMMIT;
