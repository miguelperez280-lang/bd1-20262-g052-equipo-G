-- 2.- Darle un nombre "coherente"
-- 3.- Agregar campos coherentes con la tabla
-- 4.- Realizar todas las operaciones que se solicitan a continuación
--

-- 1.3.1
-- Crear una tabla "nueva" de su iniciativa (una tabla coherente con el sistema con su nombre, no coloque "nueva" como nombre)
--
CREATE TABLE inscripcion_evento (
    campo_temporal INTEGER
);



-- 1.3.2
-- Agregar una clave primaria y otros 3 campos cualquiera a la tabla "nueva"
-- Mínimo un campo tipo texto y uno numérico
--

ALTER TABLE inscripcion_evento
ADD COLUMN id_inscripcion INTEGER GENERATED BY DEFAULT AS IDENTITY;

ALTER TABLE inscripcion_evento
ADD CONSTRAINT pk_inscripcion_evento
PRIMARY KEY (id_inscripcion);

ALTER TABLE inscripcion_evento
ADD COLUMN id_usuario INTEGER;

ALTER TABLE inscripcion_evento
ADD COLUMN id_evento INTEGER;

ALTER TABLE inscripcion_evento
ADD COLUMN estado_inscripcion VARCHAR(50);

ALTER TABLE inscripcion_evento
ADD COLUMN costo_inscripcion NUMERIC(12,2);

ALTER TABLE inscripcion_evento
ADD CONSTRAINT fk_inscripcion_evento_usuario
FOREIGN KEY (id_usuario) REFERENCES usuario (id_usuario);

ALTER TABLE inscripcion_evento
ADD CONSTRAINT fk_inscripcion_evento_evento

-- 1.3.3
-- Quitar uno de los campos de la tabla "nueva"
--

-- 1.3.4
-- Cambiar el nombre de la tabla "nueva" a otro nombre "otro_nombre"
-- Todas las operaciones siguientes se realizan sobre la tabla renombrada
--

ALTER TABLE inscripcion_evento
RENAME TO registro_inscripcion;


-- 1.3.5 
-- Agregar un campo único a la tabla 
--
ALTER TABLE registro_inscripcion

-- 1.3.6
-- Agregar 2 fechas de inicio y fin; y colocar un control de orden de fechas
--
ALTER TABLE registro_inscripcion
ADD COLUMN fecha_apertura DATE;

ALTER TABLE registro_inscripcion
ADD COLUMN fecha_cierre DATE;

ALTER TABLE registro_inscripcion
ADD CONSTRAINT ck_registro_inscripcion_fechas
CHECK (
    fecha_apertura IS NULL
    OR fecha_cierre IS NULL
    OR fecha_cierre >= fecha_apertura
);

-- 1.3.7
-- Agregar 1 campo entero y colocar un control para que no sea negativo
--
ALTER TABLE registro_inscripcion
ADD COLUMN cantidad_acompanantes INTEGER;

ALTER TABLE registro_inscripcion
ADD CONSTRAINT ck_registro_inscripcion_acompanantes

-- 1.3.8
